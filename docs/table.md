# Table

Data table component.

## Import

```tsx
import { Table, TableHeader, TableBody, TableRow, TableCell } from '@flavioespinoza/salsa-ui'
```

## Usage

```tsx
<Table>
  <TableHeader>
    <TableRow>
      <TableHead>Name</TableHead>
    </TableRow>
  </TableHeader>
  <TableBody>
    <TableRow>
      <TableCell>John</TableCell>
    </TableRow>
  </TableBody>
</Table>
```
