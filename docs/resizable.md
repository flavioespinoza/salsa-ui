# Resizable

Resizable panel layout.

## Import

```tsx
import { ResizablePanelGroup, ResizablePanel, ResizableHandle } from '@flavioespinoza/salsa-ui'
```

## Usage

```tsx
<ResizablePanelGroup direction="horizontal">
  <ResizablePanel>Panel 1</ResizablePanel>
  <ResizableHandle />
  <ResizablePanel>Panel 2</ResizablePanel>
</ResizablePanelGroup>
```
